// Add velocity checker
export async function checkVelocityLimit(
  businessId: string
): Promise<{ allowed: boolean; recentCount: number }> {
  const oneMinuteAgo = new Date(Date.now() - 60000);
  
  const recentCount = await WalletTransaction.countDocuments({
    business: businessId,
    type: TransactionType.DEBIT,
    createdAt: { $gte: oneMinuteAgo },
    status: { $in: [TransactionStatus.PENDING, TransactionStatus.SUCCESS] }
  });
  
  const VELOCITY_LIMIT = 10;  // Max 10 debits per minute
  
  return {
    allowed: recentCount < VELOCITY_LIMIT,
    recentCount
  };
}
